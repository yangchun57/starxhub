  label="选择城市"
  :width="220"
/>

<!-- 原生静态结构（展开态用 --inline 变体） -->
<div class="hx-select-wrap" style="width:220px">
  <button class="hx-select hx-select-trigger">…</button>
  <span class="hx-select-arrow">…</span>
  <div class="hx-select-panel hx-select-panel--inline">…</div>
</div>
浮层挂到 body + fixed 定位，按触发器位置计算 left/top，下方空间不足时自动向上翻转，避免被卡片 overflow:hidden 裁切。键盘：↑↓ 切换，Enter 选中，Esc 关闭。
